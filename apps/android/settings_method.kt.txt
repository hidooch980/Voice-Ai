    /** Called by MainRepository from the ViewModel-owned IO initialization job. */
    fun ensureCommunitySubscriptions() {
        // Different fallback results mean this setting has never been stored.
        // Enable telemetry on first install, preserving any explicit choice on upgrade.
        if (MmkvManager.decodeSettingsBool(AppConfig.PREF_SPEED_ENABLED, true) !=
            MmkvManager.decodeSettingsBool(AppConfig.PREF_SPEED_ENABLED, false)) {
            MmkvManager.encodeSettings(AppConfig.PREF_SPEED_ENABLED, true)
        }
        CommunitySubscriptions.installOnce(object : CommunitySubscriptions.Store {
            override fun isInstalled() = MmkvManager.decodeSettingsBool(CommunitySubscriptions.INSTALLED_KEY, false)
            override fun isLegacyInstalled() = MmkvManager.decodeSettingsBool(CommunitySubscriptions.LEGACY_INSTALLED_KEY, false)
            override fun current() = MmkvManager.decodeSubscriptions().associate { it.guid to it.subscription }
            override fun put(id: String, item: SubscriptionItem) {
                MmkvManager.encodeSubscription(id, item)
                check(MmkvManager.decodeSubscription(id) == item && id in MmkvManager.decodeSubsList()) {
                    "Community subscription storage failed"
                }
            }
            override fun remove(id: String) { MmkvManager.removeSubscription(id) }
            override fun markInstalled() {
                check(MmkvManager.encodeSettings(CommunitySubscriptions.INSTALLED_KEY, true)) {
                    "Community subscription marker failed"
                }
            }
        })
    }

    /** Bound imported preferences as well as settings edited through the UI. */
    internal fun boundedRealPingConcurrency(value: String?): Int =
        (value?.toIntOrNull() ?: 16).coerceIn(1, 64)
